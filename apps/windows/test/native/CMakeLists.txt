cmake_minimum_required(VERSION 3.14)
project(accessibility_alert_test LANGUAGES CXX)
enable_testing()
add_executable(accessibility_alert_test accessibility_alert_test.cpp)
target_compile_features(accessibility_alert_test PRIVATE cxx_std_17)
target_compile_definitions(accessibility_alert_test PRIVATE NOMINMAX)
target_compile_options(accessibility_alert_test PRIVATE /W4 /WX /EHsc)
target_link_libraries(accessibility_alert_test PRIVATE oleacc oleaut32 ole32 user32)
add_test(NAME accessibility_alert COMMAND accessibility_alert_test)
